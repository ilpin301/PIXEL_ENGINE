cmake_minimum_required(VERSION 3.20)
# PGE as the top-level project or added by a game with add_subdirectory() (S9.1). PROJECT_IS_TOP_LEVEL needs 3.21.
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
  set(PGE_TOP_LEVEL ON)
else()
  set(PGE_TOP_LEVEL OFF)
endif()
# a parent's normal variable (e.g. set(PGE_PLATFORM ...)) must survive the set(... CACHE ...) below
if(POLICY CMP0126)
  cmake_policy(SET CMP0126 NEW)
endif()

set(PGE_PLATFORM host CACHE STRING "host | stm32f429i_disco | tier B (compile-check only): rp2040 | rp2350 | stm32h7 | stm32f4_generic | esp32")
if(PGE_PLATFORM MATCHES "^(rp2040|rp2350|stm32h7|stm32f4_generic|esp32)$")
  include(cmake/tier_b.cmake)
endif()
if(PGE_TIERB)
  set(PGE_LANGS C CXX)
  if(PGE_TOP_LEVEL AND NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
  endif()
elseif(PGE_PLATFORM STREQUAL "stm32f429i_disco")
  set(PGE_LANGS CXX C ASM)
  if(PGE_TOP_LEVEL AND NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
  endif()
else()
  set(PGE_LANGS CXX)
endif()
project(pge LANGUAGES ${PGE_LANGS})

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# OFF by default when a game adds PGE with add_subdirectory()
option(PGE_BUILD_EXAMPLES "Build the engine examples, benches and example firmwares" ${PGE_TOP_LEVEL})
option(PGE_BUILD_TESTS "Build host unit tests" ${PGE_TOP_LEVEL})

if(NOT PGE_TIERB)
  set(PGE_PROFILE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/platform/${PGE_PLATFORM})
endif()
add_subdirectory(core)

if(PGE_TIERB)
  add_subdirectory(platform/tier_b)
elseif(PGE_PLATFORM STREQUAL "stm32f429i_disco")
  if(PGE_BUILD_EXAMPLES)
    # pge_b1 only; its host exe does not build for ARM, hence EXCLUDE_FROM_ALL (the scene exe depends on pge_b1).
    add_subdirectory(bench/b1 EXCLUDE_FROM_ALL)
  endif()
  add_subdirectory(platform/stm32f429i_disco)
else()

  find_package(SDL2 CONFIG QUIET)
  add_subdirectory(platform/host)
  if(PGE_BUILD_EXAMPLES)
    if(SDL2_FOUND)
      add_subdirectory(examples/proto)
      add_subdirectory(examples/gfxdemo)
      add_subdirectory(examples/scroller)
      add_subdirectory(examples/platformer)
      add_subdirectory(examples/sandbox)
      add_subdirectory(examples/planet)
      add_subdirectory(examples/audiodemo)
      add_subdirectory(examples/tutorial)
    else()
      message(STATUS "SDL2 not found: host window demo skipped")
    endif()

    add_subdirectory(examples/empty)
  endif()

  if(PGE_BUILD_EXAMPLES OR PGE_BUILD_TESTS)
    add_subdirectory(bench/b1)   # test_bench links pge_b1
  endif()

  if(PGE_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
  endif()
endif()
